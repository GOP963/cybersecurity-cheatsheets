```c++
#include <windows.h>
#include "resource.h"

HINSTANCE hInst;

LRESULT CALLBACK WndProc(HWND hWnd, UINT msg, WPARAM wParam, LPARAM lParam);
INT_PTR CALLBACK BasicDialog(HWND hDlg, UINT message, WPARAM wParam, LPARAM lParam);

struct SearchData
{
    LPCWSTR searchText;
    HWND    foundWindow;
};

BOOL CALLBACK EnumWindowsProc(HWND hwnd, LPARAM lParam)
{
    SearchData* data = (SearchData*)lParam;

    if (!IsWindowVisible(hwnd))
        return TRUE;

    wchar_t title[256] = { 0 };
    GetWindowText(hwnd, title, 256);

    if (wcslen(title) == 0)
        return TRUE;
    wchar_t titleLower[256], searchLower[256];
    lstrcpyW(titleLower, title);
    lstrcpyW(searchLower, data->searchText);
    CharLowerW(titleLower);
    CharLowerW(searchLower);

    if (wcsstr(titleLower, searchLower) != NULL)
    {
        data->foundWindow = hwnd;
        return FALSE;
    }
    return TRUE;
}
BOOL TID_PID(LPCWSTR searchText, DWORD* pid, DWORD* tid)
{
    SearchData data = { searchText, NULL };
    EnumWindows(EnumWindowsProc, (LPARAM)&data);

    if (data.foundWindow == NULL)
        return FALSE;

    *tid = GetWindowThreadProcessId(data.foundWindow, pid);
    return TRUE;
}

int WINAPI wWinMain(
    _In_ HINSTANCE hInstance,
    _In_opt_ HINSTANCE hPrevInstance,
    _In_ LPWSTR lpCmdLine,
    _In_ int nCmdShow
)
{
    hInst = hInstance;

    WNDCLASS wc = { 0 };
    wc.lpfnWndProc = WndProc;
    wc.hInstance = hInstance;
    wc.lpszClassName = L"Custom Window Class";
    wc.hbrBackground = (HBRUSH)(COLOR_WINDOW + 1);
    wc.lpszMenuName = MAKEINTRESOURCE(IDR_MENU1);

    if (!RegisterClass(&wc))
    {
        MessageBox(NULL, L"Failed to register window class!", L"Error", MB_ICONERROR | MB_OK);
        return 0;
    }

    HWND hWnd = CreateWindow(
        wc.lpszClassName,
        L"Get PID/TID Tool",
        WS_OVERLAPPEDWINDOW,
        CW_USEDEFAULT, CW_USEDEFAULT,
        800, 600,
        NULL, NULL, hInstance, NULL
    );

    if (!hWnd)
    {
        MessageBox(NULL, L"Failed to create window!", L"Error", MB_ICONERROR | MB_OK);
        return 0;
    }

    ShowWindow(hWnd, nCmdShow);
    UpdateWindow(hWnd);

    MSG msg = { 0 };
    while (GetMessage(&msg, NULL, 0, 0))
    {
        TranslateMessage(&msg);
        DispatchMessage(&msg);
    }
    return (int)msg.wParam;
}

LRESULT CALLBACK WndProc(HWND hWnd, UINT msg, WPARAM wParam, LPARAM lParam)
{
    switch (msg)
    {
    case WM_COMMAND:
        switch (LOWORD(wParam))
        {
        case ID_TOOLS_GETPIDTID:
            DialogBox(hInst, MAKEINTRESOURCE(IDD_DIALOG1), hWnd, BasicDialog);
            break;
        }
        break;

    case WM_CLOSE:
        DestroyWindow(hWnd);
        break;

    case WM_DESTROY:
        PostQuitMessage(0);
        break;

    default:
        return DefWindowProc(hWnd, msg, wParam, lParam);
    }
    return 0;
}

INT_PTR CALLBACK BasicDialog(HWND hDlg, UINT message, WPARAM wParam, LPARAM lParam)
{
    UNREFERENCED_PARAMETER(lParam);
    switch (message)
    {
    case WM_INITDIALOG:
        return (INT_PTR)TRUE;

    case WM_COMMAND:
        switch (LOWORD(wParam))
        {
        case IDOK:
        {
            wchar_t searchText[256] = { 0 };
            GetDlgItemText(hDlg, IDC_EDIT2, searchText, 256);

            if (wcslen(searchText) == 0)
            {
                MessageBox(hDlg, L"Please enter a window name", L"Error", MB_ICONWARNING | MB_OK);
                return (INT_PTR)TRUE;
            }

            DWORD pid = 0, tid = 0;
            if (TID_PID(searchText, &pid, &tid))
            {
                wchar_t buf[32];
                wsprintf(buf, L"%lu", pid);
                SetDlgItemText(hDlg, IDC_EDIT_PID, buf);

                wsprintf(buf, L"%lu", tid);
                SetDlgItemText(hDlg, IDC_EDIT_TID, buf);
            }
            else
            {
                MessageBox(hDlg, L"Window not found!", L"Error", MB_ICONWARNING | MB_OK);
                SetDlgItemText(hDlg, IDC_EDIT_PID, L"");
                SetDlgItemText(hDlg, IDC_EDIT_TID, L"");
            }
            return (INT_PTR)TRUE;
        }

        case IDCANCEL:
            EndDialog(hDlg, IDCANCEL);
            return (INT_PTR)TRUE;
        }
        break;
    }
    return (INT_PTR)FALSE;
}
```

3 ذخيره وبازيابى موقعيت واندازه پنجره برنامه وظايف: برنامه Notepad را باز كنيد. موقعيت واندازه ينجره را تغيير دهيد. برنامه را ببنديد. در اجراى بعدى، Notepad بايد در همان موقعيت واندازه اى كه بسته شده بود باز شود. يك پروزه Windows Desktop Application ايجاد كنيد. موقعيت واندازه پنجره را در فايل يا رجيسترى ذخيره كنيد. در هنكام باز شدن برنامه، اطلاعات ذخيرهشده را از فايل يا رجيسترى بازيابى كنيد و در همان موقعيت واندازه نمايش داده شود

## این "cache شدن" خودکار نیست!

ویندوز به صورت پیش‌فرض **هیچ** اطلاعاتی از اندازه/موقعیت پنجره‌ی برنامه‌ها رو ذخیره نمی‌کنه. این خود Notepad (و هر برنامه‌ی دیگه‌ای که این رفتار رو داره) هست که وقتی می‌بندیش، به صورت دستی این اطلاعات رو یه جایی ذخیره می‌کنه و وقتی دوباره باز میشه، خودش میاد این مقادیر رو می‌خونه و پنجره‌ش رو با همون سایز/موقعیت می‌سازه.

Notepad (مثل اکثر برنامه‌های ویندوزی) این اطلاعات رو تو **Registry** ذخیره می‌کنه، مثلاً یه جایی شبیه:

```
HKEY_CURRENT_USER\Software\Microsoft\Notepad
```

اگه بخوای، می‌تونی با `regedit` بری همینجا رو نگاه کنی و ببینی مقادیری مثل `iWindowPosX`, `iWindowPosY`, `iWindowPosDX`, `iWindowPosDY` رو ذخیره کرده.

## API های کلیدی که برای تمرین بعدی لازم داری

### ۱. برای خوندن موقعیت/اندازه‌ی فعلی پنجره (قبل از بستن)

```cpp
GetWindowPlacement(HWND hWnd, WINDOWPLACEMENT* lpwndpl);
```

این تابع یک struct به اسم `WINDOWPLACEMENT` رو پر می‌کنه که شامل موقعیت، اندازه، و حالت پنجره (Normal/Maximized/Minimized) هست — خیلی کامل‌تر از اینه که فقط `GetWindowRect` بگیری.

### ۲. برای اعمال دوباره‌ی موقعیت/اندازه‌ی ذخیره‌شده (موقع باز کردن)

```cpp
SetWindowPlacement(HWND hWnd, const WINDOWPLACEMENT* lpwndpl);
```

### ۳. برای ذخیره در Registry

```cpp
RegCreateKeyEx / RegSetValueEx / RegOpenKeyEx / RegQueryValueEx / RegCloseKey
```

### یا به جای Registry، می‌تونی در یک فایل ساده ذخیره کنی:

```cpp
CreateFile / WriteFile / ReadFile
```

(یا حتی ساده‌تر: `WritePrivateProfileString` و `GetPrivateProfileString` برای فایل‌های `.ini`)

## نکته‌ی مهم برای معماری این تمرین

چون تمرینت میگه **"برنامه Notepad رو باز کنید"**، احتمالاً باید:

1. برنامه‌ت (نه Notepad) خودش با `CreateProcess` نوت‌پد رو باز کنه.
2. موقعیت/اندازه‌ی پنجره‌ی Notepad (که پیدا می‌کنی، دقیقاً مثل تمرین قبلی با `FindWindow`) رو بخونه و وقتی می‌بندیش ذخیره کنه (یا این کارو موقع بسته شدن برنامه‌ی خودت انجام بدی).
3. دفعه‌ی بعد که برنامه‌ت اجرا میشه، دوباره Notepad رو با `CreateProcess` باز کنه و بلافاصله با `SetWindowPlacement` موقعیت/اندازه‌ی ذخیره‌شده رو روش اعمال کنه.
